def encode(name, raw, bits, channels, block, label, exe):
    source = work / (name + '.raw')
    source.write_bytes(raw)
    dst = work / (name + '-' + label + '.flac')
    args = [str(exe), 'encode', str(source), str(dst), '--raw', '--sample-rate', '48000',
            '--channels', str(channels), '--bits', str(bits), '--effort', '8', '--block-size', str(block)]
    p = subprocess.run(args, capture_output=True, timeout=60)
    assert p.returncode == 0, (args, p.returncode, p.stderr)
    return dst

def check(name, raw, bits, channels, block, exact=False, prior=False):
    width = (bits + 7) // 8
    values = [int.from_bytes(raw[i:i+width], 'little', signed=True) for i in range(0, len(raw), width)]
    paths = {}
    for label, exe in [('baseline', out/'baseline.exe'), ('candidate', out/'candidate.exe'), ('fresh', fresh)]:
        paths[label] = encode(name, raw, bits, channels, block, label, exe)
    data = {k: p.read_bytes() for k, p in paths.items()}
    assert data['candidate'] == data['fresh'], name
    if exact:
        assert data['candidate'] == data['baseline'], ('baseline mismatch', name)
    if prior:
        old = encode(name, raw, bits, channels, block, 'revision3', previous/'candidate.exe')
        assert old.read_bytes() == data['candidate'], ('long strategy changed', name)
    props, pcm = ref.decode(paths['fresh'])
    assert props == (channels, bits, 48000) and pcm == values, name
    assert data['fresh'][26:42] == hashlib.md5(raw).digest(), name
    p = subprocess.run([str(flac), '--silent', '-t', str(paths['fresh'])], capture_output=True, timeout=60)
    assert p.returncode == 0, (name, p.stderr)
    records.append(dict(name=name, bits=bits, channels=channels, frames=len(values)//channels,
                        block=block, baseline_bytes=len(data['baseline']), candidate_bytes=len(data['candidate']),
                        exact_baseline_required=exact, unchanged_revision3_required=prior,
                        libflac_pcm='exact', md5='exact', flac_test_exit=0))
    return paths

rng = random.Random(202609291339)
def generated(frames, bits, channels):
    lo, hi = -(1 << (bits-1)), (1 << (bits-1))-1
    values = [rng.choice([lo, hi]) if (i+c*19)%113 == 0 else ((i+c)%5)-2
              for i in range(frames) for c in range(channels)]
    return b''.join(v.to_bytes((bits+7)//8, 'little', signed=True) for v in values)

def last_frame(path):
    ana = path.with_suffix('.ana')
    p = subprocess.run([str(flac), '--silent', '-a', '-f', '-o', str(ana), str(path)], capture_output=True, timeout=60)
    assert p.returncode == 0, p.stderr
    frames = [dict(re.findall(r'(\w+)=([0-9]+)', l)) for l in ana.read_text().splitlines() if l.startswith('frame=')]
    assert len(frames) == 2, frames
    last = frames[-1]
    start, bits = int(last['offset']), int(last['bits'])
    assert bits % 8 == 0 and start+bits//8 == path.stat().st_size
    return int(last['blocksize']), path.read_bytes()[start:]

for block, tail in [(1024, 1), (1024, 37), (4096, 255), (4096, 256)]:

    assert left[0] == right[0] == tail and left[1] == right[1], (block, tail)
