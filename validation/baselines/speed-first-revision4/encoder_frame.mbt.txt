// Exact integer frame serialization and cost-based subframe selection, RFC 9639 §9.

///|
priv struct EncoderBits {
  bytes : Array[Byte]
  mut current : Int
  mut used : Int
}

///|
fn EncoderBits::new() -> EncoderBits {
  { bytes: [], current: 0, used: 0, }
}

///|
fn EncoderBits::write(self : EncoderBits, value : Int64, width : Int) -> Unit {
  let mut remaining = width
  while remaining > 0 {
    let take = if remaining < 8 - self.used { remaining } else { 8 - self.used }
    remaining -= take
    self.current = (self.current << take) |
      ((value >> remaining).to_int() & ((1 << take) - 1))
    self.used += take
    if self.used == 8 {
      self.bytes.push(self.current.to_byte())
      self.current = 0
      self.used = 0
    }
  }
}

///|
fn EncoderBits::unary(self : EncoderBits, zeros : Int64) -> Unit {
  let mut left = zeros
  if self.used != 0 {
    let take = if left < (8 - self.used).to_int64() {
      left.to_int()
    } else {
      8 - self.used
    }
    self.write(0L, take)
    left -= take.to_int64()
  }
  while left >= 8L {
    self.bytes.push(0)
    left -= 8L
  }
  self.write(0L, left.to_int())
  self.write(1L, 1)
}

///|
fn EncoderBits::align(self : EncoderBits) -> Unit {
  if self.used != 0 {
    self.write(0L, 8 - self.used)
  }
}

///|
fn encoder_number(
  out : EncoderBits,
  number : Int64,
  variable : Bool,
) -> Unit raise FlacError {
  if number < 0L || number > (if variable { 68719476735L } else { 2147483647L }) {
    raise Invalid(0L, "encoded frame/sample number exceeds format range")
  }
  if number < 128L {
    out.write(number, 8)
    return
  }
  let limits = [2048L, 65536L, 2097152L, 67108864L, 2147483648L, 68719476736L]
  let mut count = 2
  while number >= limits[count - 2] {
    count += 1
  }
  let prefix = (256 - (1 << (8 - count))).to_int64()
  out.write(prefix | (number >> ((count - 1) * 6)), 8)
  for i = count - 2; i >= 0; i = i - 1 {
    out.write(128L | ((number >> (i * 6)) & 63L), 8)
  }
}

///|
priv struct EncoderPartition {
  parameter : Int
  width : Int
  count : Int
}

///|
priv struct EncoderResidual {
  order : Int
  rice2 : Bool
  parts : Array[EncoderPartition]
  bits : Int64
}

// Quotient sums permit all Rice parameters and partition levels to be costed
// without writing speculative unary strings. Nodes are merged bottom-up.

///|
priv struct EncoderRiceNode {
  quotients : Array[Int64]
  mut count : Int
  mut width : Int
}

///|
fn encoder_signed_width(value : Int64) -> Int {
  if value == 0L {
    return 0
  }
  // For negative values, complement removes the redundant sign extension.
  65 - (if value < 0L { value.lnot() } else { value }).clz()
}

///|
fn encoder_rice_parameter(node : EncoderRiceNode, maximum : Int) -> Int {
  // Bit lengths give a conservative lower bound within one of the original
  // repeated-doubling bound. No quotient division or approximate cost is used.
  let bound = node.count.to_int64().clz() - node.quotients[0].clz() - 2
  let mut parameter = if bound < 0 {
    0
  } else if bound > maximum {
    maximum
  } else {
    bound
  }
  // For every skipped k, sum(ceil((u >> k)/2)) > count follows
  // from sum(u) > count * 2^(k+2). Thus C(k+1) < C(k) strictly.
  // The bound is conservative for arbitrary residual distributions.
  while parameter < maximum &&
        node.quotients[0] > node.count.to_int64() << (parameter + 2) {
    parameter += 1
  }
  let mut cost = node.quotients[parameter] +
    node.count.to_int64() * (parameter + 1).to_int64()
  for k = parameter + 1; k <= maximum; k = k + 1 {
    let candidate = node.quotients[k] +
      node.count.to_int64() * (k + 1).to_int64()
    if candidate < cost {
      cost = candidate
      parameter = k
    } else {
      // C(k+1)-C(k) = count - sum(ceil((u >> k)/2)). This
      // difference is nondecreasing, so the first non-improvement is final.
      // Keeping the earlier parameter also preserves the existing tie rule.
      break
    }
  }
  parameter
}

///|
fn encoder_partition(
  node : EncoderRiceNode,
  rice2 : Bool,
) -> (EncoderPartition, Int64) {
  let parameter_bits = if rice2 { 5 } else { 4 }
  let maximum = if rice2 { 30 } else { 14 }
  let parameter = encoder_rice_parameter(node, maximum)
  let cost = node.quotients[parameter] +
    node.count.to_int64() * (parameter + 1).to_int64() +
    parameter_bits.to_int64()
  if node.width <= 31 {
    let escaped = (parameter_bits + 5).to_int64() +
      node.count.to_int64() * node.width.to_int64()
    if escaped < cost {
      return (
        {
          parameter: (1 << parameter_bits) - 1,
          width: node.width,
          count: node.count,
        },
        escaped,
      )
    }
  }
  ({ parameter, width: -1, count: node.count, }, cost)
}

///|
fn encoder_residual_plan(
  samples : Array[Int64],
  predictor : Int,
  max_partition : Int,
  workspace? : Array[Array[Int64]] = [],
) -> EncoderResidual {
  let size = samples.length()
  if predictor < size {
    let value = samples[predictor]
    let mut constant = true
    for i = predictor + 1; i < size; i = i + 1 {
      if samples[i] != value {
        constant = false
        break
      }
    }
    if constant {
      // Each partition has the same Rice slope and escape width. Splitting
      // adds parameter/escape headers and cannot improve the minimum of those
      // two affine costs. Partition zero wins all ties.
      let count = size - predictor
      let folded = if value < 0L { -value * 2L - 1L } else { value * 2L }
      let node : EncoderRiceNode = {
        quotients: Array::makei(31, k => (folded >> k) * count.to_int64()),
        count,
        width: encoder_signed_width(value),
      }
      let (part0, cost0) = encoder_partition(node, false)
      let (part1, cost1) = encoder_partition(node, true)
      return if cost1 < cost0 {
        { order: 0, rice2: true, parts: [part1], bits: 6L + cost1, }
      } else {
        { order: 0, rice2: false, parts: [part0], bits: 6L + cost0, }
      }
    }
  }
  let mut order = 0
  while order < max_partition &&
        size % (1 << (order + 1)) == 0 &&
        size >> (order + 1) > predictor {
    order += 1
  }
  let partition_size = size >> order
  let nodes : Array[EncoderRiceNode] = []
  for p = 0; p < 1 << order; p = p + 1 {
    let start = if p == 0 { predictor } else { p * partition_size }
    let end = (p + 1) * partition_size
    // Scratch belongs to one subframe search. Plans retain only scalar records,
    // never these arrays, so later candidates may clear and reuse every slot.
    if p == workspace.length() {
      workspace.push(Array::make(31, 0L))
    }
    let quotients = workspace[p]
    for k = 0; k < 31; k = k + 1 {
      quotients[k] = 0L
    }
    let mut width = 0
    for i = start; i < end; i = i + 1 {
      let value = samples[i]
      let w = encoder_signed_width(value)
      if w > width {
        width = w
      }
      let mut folded = if value < 0L { -value * 2L - 1L } else { value * 2L }
      let mut k = 0
      while folded != 0L && k <= 30 {
        quotients[k] += folded
        folded = folded >> 1
        k += 1
      }
    }
    nodes.push({ quotients, count: end - start, width, })
  }
  // Store compact scalar choices for all levels, materializing records only
  // after the global winning level is known. Total tree nodes are < 2 * leaves.
  let choices0 = Array::make(nodes.length() * 2, 0)
  let choices1 = Array::make(nodes.length() * 2, 0)
  let mut active = nodes.length()
  let mut offset = 0
  let mut best_offset = 0
  let mut best_order = 0
  let mut best_rice2 = false
  let mut best_bits = 9223372036854775807L
  while true {
    let mut bits0 = 6L
    let mut bits1 = 6L
    for i = 0; i < active; i = i + 1 {
      let node = nodes[i]
      let k1 = encoder_rice_parameter(node, 30)
      let k0 = if k1 > 14 { 14 } else { k1 }
      let count = node.count.to_int64()
      let cost0 = node.quotients[k0] + count * (k0 + 1).to_int64() + 4L
      let cost1 = node.quotients[k1] + count * (k1 + 1).to_int64() + 5L
      let escape0 = 9L + count * node.width.to_int64()
      if node.width <= 31 && escape0 < cost0 {
        choices0[offset + i] = 32 + node.width
        bits0 += escape0
      } else {
        choices0[offset + i] = k0
        bits0 += cost0
      }
      if node.width <= 31 && escape0 + 1L < cost1 {
        choices1[offset + i] = 32 + node.width
        bits1 += escape0 + 1L
      } else {
        choices1[offset + i] = k1
        bits1 += cost1
      }
    }
    // The earliest convex minimum clamped to 14 is the Rice minimum.
    // Rice wins method ties; later iterations have lower partition orders.
    let rice2 = bits1 < bits0
    let bits = if rice2 { bits1 } else { bits0 }
    if bits <= best_bits {
      best_bits = bits
      best_offset = offset
      best_order = order
      best_rice2 = rice2
    }
    offset += active
    if order == 0 {
      break
    }
    // Compact the tree in place. Every parent consumes two nodes before its
    // slot is overwritten; the quotient arrays can be reused without aliasing.
    for i = 0; i < active; i = i + 2 {
      let a = nodes[i]
      let b = nodes[i + 1]
      let width = if a.width > b.width { a.width } else { b.width }
      let columns = if width < 31 { width } else { 31 }
      // Folded residuals have at most `width` bits; higher columns are zero.
      for k = 0; k < columns; k = k + 1 {
        a.quotients[k] += b.quotients[k]
      }
      a.count += b.count
      a.width = width
      nodes[i / 2] = a
    }
    active = active / 2
    order -= 1
  }
  let choices = if best_rice2 { choices1 } else { choices0 }
  let count = size >> best_order
  let parts = Array::makei(1 << best_order, i => {
    let choice = choices[best_offset + i]
    {
      parameter: if choice >= 32 {
        if best_rice2 {
          31
        } else {
          15
        }
      } else {
        choice
      },
      width: if choice >= 32 {
        choice - 32
      } else {
        -1
      },
      count: if i == 0 {
        count - predictor
      } else {
        count
      },
    }
  })
  { order: best_order, rice2: best_rice2, parts, bits: best_bits, }
}

///|
priv struct EncoderSubframe {
  kind : Int
  wasted : Int
  depth : Int
  samples : Array[Int64]
  coefficients : Array[Int64]
  shift : Int
  precision : Int
  residual : Array[Int64]
  plan : EncoderResidual?
  bits : Int64
}

///|
fn encoder_fixed_coefficients(order : Int) -> Array[Int64] {
  match order {
    0 => []
    1 => [1L]
    2 => [2L, -1L]
    3 => [3L, -3L, 1L]
    _ => [4L, -6L, 4L, -1L]
  }
}

///|
fn encoder_predictor(
  samples : Array[Int64],
  coefficients : Array[Int64],
  shift : Int,
) -> Array[Int64]? {
  let order = coefficients.length()
  let residual = Array::make(samples.length(), 0L)
  for i = order; i < samples.length(); i = i + 1 {
    let mut prediction = 0L
    for j = 0; j < order; j = j + 1 {
      prediction += coefficients[j] * samples[i - j - 1]
    }
    let value = samples[i] - (prediction >> shift)
    if value <= -2147483648L || value >= 2147483648L {
      return None
    }
    residual[i] = value
  }
  Some(residual)
}

///|
fn encoder_subframe(
  samples : Array[Int64],
  width : Int,
  effort : Int,
) -> EncoderSubframe {
  let size = samples.length()
  let workspace : Array[Array[Int64]] = []
  let mut wasted = width - 1
  let mut constant_input = true
  for sample in samples {
    if sample != samples[0] {
      constant_input = false
    }
    if sample != 0L {
      let mut zeros = 0
      while zeros < wasted && (sample & (1L << zeros)) == 0L {
        zeros += 1
      }
      if zeros < wasted {
        wasted = zeros
      }
    }
  }
  let mut best : EncoderSubframe = {
    kind: 1,
    wasted: 0,
    depth: width,
    samples,
    coefficients: [],
    shift: 0,
    precision: 0,
    residual: [],
    plan: None,
    bits: 8L + size.to_int64() * width.to_int64(),
  }
  let choices = if wasted == 0 { [0] } else { [0, wasted] }
  for wasted in choices {
    let depth = width - wasted
    let shifted = if wasted == 0 {
      samples
    } else {
      samples.map(x => x >> wasted)
    }
    let header_bits = 8L + wasted.to_int64()
    let constant = constant_input
    if constant && header_bits + depth.to_int64() < best.bits {
      best = {
        ..best,
        kind: 0,
        wasted,
        depth,
        samples: shifted,
        bits: header_bits + depth.to_int64(),
      }
    }
    let verbatim = header_bits + size.to_int64() * depth.to_int64()
    if verbatim < best.bits {
      best = {
        ..best,
        kind: 1,
        wasted,
        depth,
        samples: shifted,
        plan: None,
        bits: verbatim,
      }
    }
    if effort == 0 {
      continue
    }
    // A constant signal can still be cheaper as fixed order zero (notably
    // wide-depth zero blocks with escape width zero). Higher predictors need
    // a warm-up at least as expensive as the entire constant sample field.
    let max_partition = if constant {
      0
    } else if effort == 8 {
      ENCODER_PARTITION_CAP
    } else {
      effort
    }
    let max_fixed = if constant { 0 } else { 4 }
    for order = 0; order <= max_fixed && order < size; order = order + 1 {
      let coefficients = encoder_fixed_coefficients(order)
      if encoder_predictor(shifted, coefficients, 0) is Some(residual) {
        let plan = encoder_residual_plan(
          residual,
          order,
          if !constant && effort == 8 {
            encoder_sparse_partition_limit(residual, order)
          } else {
            max_partition
          },
          workspace~,
        )
        let cost = header_bits + (order * depth).to_int64() + plan.bits
        if cost < best.bits {
          best = {
            kind: order + 8,
            wasted,
            depth,
            samples: shifted,
            coefficients,
            shift: 0,
            precision: 0,
            residual,
            plan: Some(plan),
            bits: cost,
          }
        }
      }
    }
    if constant {
      continue
    }
    let limits = [0, 0, 2, 4, 8, 12, 16, 24, 32]
    let max_order = if limits[effort] < size {
      limits[effort]
    } else {
      size - 1
    }
    let models = encoder_lpc_models(shifted, max_order)
    // A short block already fits within the sampling budget. Evaluate its
    // complete model pool in original order: a width proxy can badly rank
    // shared Rice costs, and offers no sampling reduction at this size.
    let shortlist = effort == 8 && size > ENCODER_LPC_PROBES
    let candidates = if shortlist {
      // Rank the full pool so an illegal full residual does not consume a
      // finalist slot. Sampled legality cannot establish whole-frame legality.
      encoder_lpc_shortlist(shifted, models, depth, limit=models.length())
    } else {
      models
    }
    let mut valid_candidates = 0
    for model in candidates {
      if shortlist && valid_candidates >= ENCODER_LPC_SHORTLIST {
        break
      }
      let order = model.coefficients.length()
      if order > 0 &&
        order <= max_order &&
        model.shift >= 0 &&
        model.shift <= 15 &&
        model.precision >= 1 &&
        model.precision <= 15 {
        if encoder_predictor(shifted, model.coefficients, model.shift)
          is Some(residual) {
          valid_candidates += 1
          let plan = encoder_residual_plan(
            residual,
            order,
            if effort == 8 {
              encoder_sparse_partition_limit(residual, order)
            } else {
              max_partition
            },
            workspace~,
          )
          let cost = header_bits +
            (order * depth + 9 + order * model.precision).to_int64() +
            plan.bits
          if cost < best.bits {
            best = {
              kind: order + 31,
              wasted,
              depth,
              samples: shifted,
              coefficients: model.coefficients,
              shift: model.shift,
              precision: model.precision,
              residual,
              plan: Some(plan),
              bits: cost,
            }
          }
        }
      }
    }
  }
  best
}

///|
fn encoder_write_subframe(out : EncoderBits, sub : EncoderSubframe) -> Unit {
  out.write(
    ((sub.kind << 1) | (if sub.wasted > 0 { 1 } else { 0 })).to_int64(),
    8,
  )
  if sub.wasted > 0 {
    out.unary((sub.wasted - 1).to_int64())
  }
  if sub.kind == 0 {
    out.write(sub.samples[0], sub.depth)
    return
  }
  if sub.kind == 1 {
    for sample in sub.samples {
      out.write(sample, sub.depth)
    }
    return
  }
  let order = sub.coefficients.length()
  for i = 0; i < order; i = i + 1 {
    out.write(sub.samples[i], sub.depth)
  }
  if sub.kind >= 32 {
    out.write((sub.precision - 1).to_int64(), 4)
    out.write(sub.shift.to_int64(), 5)
    for coefficient in sub.coefficients {
      out.write(coefficient, sub.precision)
    }
  }
  if sub.plan is Some(plan) {
    out.write(if plan.rice2 { 1L } else { 0L }, 2)
    out.write(plan.order.to_int64(), 4)
    let mut cursor = order
    for part in plan.parts {
      out.write(part.parameter.to_int64(), if plan.rice2 { 5 } else { 4 })
      if part.width >= 0 {
        out.write(part.width.to_int64(), 5)
      }
      for i = 0; i < part.count; i = i + 1 {
        let value = sub.residual[cursor]
        if part.width >= 0 {
          out.write(value, part.width)
        } else {
          let folded = if value < 0L { -value * 2L - 1L } else { value * 2L }
          out.unary(folded >> part.parameter)
          out.write(folded, part.parameter)
        }
        cursor += 1
      }
    }
  }
}

///|
fn encode_frame(
  format : PcmFormat,
  planes : Array[Array[Int]],
  number : Int64,
  variable_block : Bool,
  effort : Int,
) -> Bytes raise FlacError {
  if format.channels < 1 ||
    format.channels > 8 ||
    format.bits_per_sample < 4 ||
    format.bits_per_sample > 32 ||
    format.sample_rate < 0 ||
    format.sample_rate > 1048575 ||
    planes.length() != format.channels ||
    effort < 0 ||
    effort > 8 {
    raise Invalid(0L, "invalid encoder frame format or effort")
  }
  let size = planes[0].length()
  if size < 1 || size > 65535 {
    raise Invalid(0L, "encoder block size must be 1..65535")
  }
  let bound = 1L << (format.bits_per_sample - 1)
  let original : Array[Array[Int64]] = []
  for plane in planes {
    if plane.length() != size {
      raise Invalid(0L, "encoder channel lengths differ")
    }
    for value in plane {
      if value.to_int64() < -bound || value.to_int64() >= bound {
        raise Invalid(0L, "input PCM exceeds declared bit depth")
      }
    }
    original.push(plane.map(x => x.to_int64()))
  }
  let mut assignment = format.channels - 1
  let mut chosen = original.map(samples => {
    encoder_subframe(samples, format.bits_per_sample, effort)
  })
  let mut cost = 0L
  for sub in chosen {
    cost += sub.bits
  }
  if format.channels == 2 && effort > 0 {
    let left = original[0]
    let right = original[1]
    let side = Array::makei(size, i => left[i] - right[i])
    let side_sub = encoder_subframe(side, format.bits_per_sample + 1, effort)
    let mid = Array::makei(size, i => (left[i] + right[i]) >> 1)
    let mid_sub = encoder_subframe(mid, format.bits_per_sample, effort)
    let combinations = [
      [chosen[0], side_sub],
      [side_sub, chosen[1]],
      [mid_sub, side_sub],
    ]
    for i = 0; i < 3; i = i + 1 {
      let candidate = combinations[i]
      let bits = candidate[0].bits + candidate[1].bits
      if bits < cost {
        assignment = i + 8
        chosen = candidate
        cost = bits
      }
    }
  }
  let out = EncoderBits::new()
  out.write(if variable_block { 0xFFF9L } else { 0xFFF8L }, 16)
  out.write(if size <= 256 { 6L } else { 7L }, 4)
  out.write(0L, 4) // STREAMINFO carries the full 20-bit sample-rate domain.
  out.write(assignment.to_int64(), 4)
  let depth_code = match format.bits_per_sample {
    8 => 1
    12 => 2
    16 => 4
    20 => 5
    24 => 6
    32 => 7
    _ => 0
  }
  out.write(depth_code.to_int64(), 3)
  out.write(0L, 1)
  encoder_number(out, number, variable_block)
  out.write((size - 1).to_int64(), if size <= 256 { 8 } else { 16 })
  let header = Bytes::from_array(out.bytes)
  out.write(frame_crc(header, 0, header.length(), 8).to_int64(), 8)
  for sub in chosen {
    encoder_write_subframe(out, sub)
  }
  out.align()
  let body = Bytes::from_array(out.bytes)
  out.write(frame_crc(body, 0, body.length(), 16).to_int64(), 16)
  Bytes::from_array(out.bytes)
}
