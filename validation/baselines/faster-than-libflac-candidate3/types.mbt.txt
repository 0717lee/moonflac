///|
/// Positioned format errors, incomplete input, and explicit resource limits.
pub suberror FlacError {
  NeedMore
  Invalid(Int64, String)
  Limit(String)
} derive(Show)

///|
/// STREAMINFO values. A zero total_samples or all-zero MD5 means unknown.
pub(all) struct StreamInfo {
  min_block_size : Int
  max_block_size : Int
  min_frame_size : Int
  max_frame_size : Int
  sample_rate : Int
  channels : Int
  bits_per_sample : Int
  total_samples : Int64
  md5 : Bytes
} derive(Show)

///|
/// Owned planar signed PCM, unshifted at its original bit depth.
pub(all) struct Frame {
  sample_number : Int64
  number : Int64
  variable_block : Bool
  sample_rate : Int
  bits_per_sample : Int
  channels : Array[Array[Int]]
  block_size : Int
  byte_offset : Int64
  byte_length : Int
} derive(Show)
