///|
/// Incremental RFC 1321 state; only one partial block is retained.
struct Md5 {
  h : Array[UInt]
  pending : Array[Byte]
  words : Array[UInt]
  mut length : Int64
}

///|
fn Md5::new() -> Md5 {
  {
    h: [0x67452301U, 0xefcdab89U, 0x98badcfeU, 0x10325476U],
    pending: [],
    words: Array::make(16, 0U),
    length: 0L,
  }
}

///|
fn Md5::block(self : Md5, bytes : Array[Byte]) -> Unit {
  let words = self.words
  for i in 0..<16 {
    let j = i * 4
    words[i] = bytes[j].to_uint() |
      (bytes[j + 1].to_uint() << 8) |
      (bytes[j + 2].to_uint() << 16) |
      (bytes[j + 3].to_uint() << 24)
  }
  let mut a = self.h[0]
  let mut b = self.h[1]
  let mut c = self.h[2]
  let mut d = self.h[3]
  for i in 0..<64 {
    let (f, g) = if i < 16 {
      ((b & c) | (b.lnot() & d), i)
    } else if i < 32 {
      ((d & b) | (d.lnot() & c), (5 * i + 1) % 16)
    } else if i < 48 {
      (b ^ c ^ d, (3 * i + 5) % 16)
    } else {
      (c ^ (b | d.lnot()), 7 * i % 16)
    }
    let value = a + f + md5_constants[i] + words[g]
    let shift = md5_shifts[i / 16 * 4 + i % 4]
    let rotated = (value << shift) | (value >> (32 - shift))
    a = d
    d = c
    c = b
    b = b + rotated
  }
  self.h[0] += a
  self.h[1] += b
  self.h[2] += c
  self.h[3] += d
}

///|
fn Md5::update(self : Md5, bytes : Bytes) -> Unit {
  self.length += bytes.length().to_int64()
  for byte in bytes {
    self.pending.push(byte)
    if self.pending.length() == 64 {
      self.block(self.pending)
      self.pending.clear()
    }
  }
}

///|
/// Finalization operates on a copy, so callers may keep appending afterward.
fn Md5::digest(self : Md5) -> Bytes {
  let copy : Md5 = {
    h: self.h.copy(),
    pending: self.pending.copy(),
    words: Array::make(16, 0U),
    length: self.length,
  }
  let bit_length = self.length << 3
  let padding = if self.pending.length() < 56 {
    56 - self.pending.length()
  } else {
    120 - self.pending.length()
  }
  copy.update(Bytes::makei(padding, i => if i == 0 { 0x80 } else { 0 }))
  copy.update(
    Bytes::makei(8, i => ((bit_length >> (i * 8)) & 255L).to_int().to_byte()),
  )
  Bytes::makei(16, i => ((copy.h[i / 4] >> (i % 4 * 8)) & 255U).to_byte())
}

///|
fn md5_update_frame(state : Md5, frame : Frame) -> Unit {
  let width = (frame.bits_per_sample + 7) / 8
  state.length += (frame.block_size * frame.channels.length() * width).to_int64()
  for i = 0; i < frame.block_size; i = i + 1 {
    for channel in frame.channels {
      let sample = channel[i]
      for byte = 0; byte < width; byte = byte + 1 {
        state.pending.push(((sample >> (byte * 8)) & 255).to_byte())
        if state.pending.length() == 64 {
          state.block(state.pending)
          state.pending.clear()
        }
      }
    }
  }
}
